✅ Шаг 1: Скачать образ Redis
bash
docker pull redis
✅ Шаг 2: Запустить контейнер с Redis
bash
docker run --name my-redis1 -d redis
✅ Шаг 3: Посмотреть список контейнеров
bash
docker ps
✅ Шаг 4: Остановить контейнер
bash
docker stop my-redis1
✅ Шаг 5: Запустить контейнер снова
bash
docker start my-redis1
✅ Шаг 6: Подключиться к Redis и продемонстрировать работу
bash
# Подключиться к клиенту Redis
docker exec -it my-redis1 redis-cli

# Внутри redis-cli выполнить:
PING           # Проверить подключение - должен ответить PONG
SET key1 "Hello Docker"  # Создать ключ
GET key1       # Получить значение
KEYS *         # Показать все ключи (аналог "списка таблиц")
INFO server    # Показать информацию о сервере
EXIT           # Выйти

bash

docker run --name my-redis3 -d redis

docker ps
✅ Шаг 8: Подключиться к контейнеру и поработать с файлами
bash

# Внутри контейнера выполнить:
pwd                    # Показать текущую директорию
ls -la /               # Посмотреть список файлов в корне
echo "test" > /tmp/test.txt  # Создать файл
cat /tmp/test.txt      # Проверить содержимое
ls -la /data           # Посмотреть данные Redis (если есть)
rm /tmp/test.txt       # Удалить файл
ls /tmp/               # Проверить, что файл удалён
exit                   # Выйти из контейнера
📝 Полный скрипт для выполнения всего задания:
bash
#!/bin/bash
echo "=== Начало выполнения задания с Redis ==="

echo "1. Скачиваем образ Redis..."
docker pull redis

echo -e "\n2. Запускаем первый контейнер..."
docker run --name redis1 -d redis

echo -e "\n3. Смотрим список контейнеров:"
docker ps

echo -e "\n4. Останавливаем контейнер..."
docker stop redis1

echo -e "\n5. Запускаем контейнер снова..."
docker start redis1

echo -e "\n6. Подключаемся к Redis и демонстрируем работу:"
echo "=== Работа с Redis ==="
docker exec redis1 redis-cli PING
docker exec redis1 redis-cli SET test_key "Hello World"
docker exec redis1 redis-cli GET test_key
docker exec redis1 redis-cli KEYS '*'

echo -e "\n7. Запускаем ещё 2 контейнера..."
docker run --name redis2 -d redis
docker run --name redis3 -d redis

echo -e "\n8. Проверяем все 3 контейнера:"
docker ps

echo -e "\n9. Подключаемся к контейнеру и работаем с файлами:"
echo "=== Работа с файлами внутри контейнера ==="
docker exec redis1 bash -c "echo 'Файл создан внутри контейнера' > /test.txt"
docker exec redis1 bash -c "cat /test.txt"
docker exec redis1 bash -c "ls -la / | grep test.txt"
docker exec redis1 bash -c "rm /test.txt"
docker exec redis1 bash -c "echo 'Файл удалён, проверяем:' && ls /test.txt 2>/dev/null || echo 'Файл не найден'"

echo -e "\n10. Очистка (опционально):"
echo "Для удаления контейнеров выполните:"
echo "docker stop redis1 redis2 redis3"
echo "docker rm redis1 redis2 redis3"
🎯 Краткая шпаргалка команд:
bash
# Основные команды
docker pull redis                  # Скачать Redis
docker run --name ИМЯ -d redis    # Запустить контейнер
docker ps                         # Показать запущенные контейнеры
docker ps -a                      # Показать все контейнеры
docker stop ИМЯ                   # Остановить контейнер
docker start ИМЯ                  # Запустить остановленный
docker exec -it ИМЯ redis-cli     # Подключиться к Redis
docker exec -it ИМЯ bash          # Подключиться к bash контейнера

# Команды внутри redis-cli
PING                              # Проверить подключение
SET ключ значение                 # Создать запись
GET ключ                          # Получить запись
KEYS *                            # Показать все ключи
INFO                              # Показать информацию
EXIT                              # Выйти

# Команды внутри bash контейнера
pwd                               # Текущая директория
ls -la                            # Список файлов
echo текст > файл.txt             # Создать файл
cat файл.txt                      # Показать файл
rm файл.txt                       # Удалить файл
